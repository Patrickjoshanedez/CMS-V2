These are critical for the UI to accurately represent what the backend computed:
// ACCURACY: Do not display `similarity` (lexical only) as the primary score.
// Always use `finalScore` which is the blended lexical + semantic result.
// See: archivePlagiarismScan.service.js:532–536

// ACCURACY: Overall similarity is character-interval based, NOT an average of source scores.
// It = matchedCharacters / totalExtractedCharacters * 100
// Do NOT recompute it from sources array — use the value from overallSimilarity directly.
// See: archivePlagiarismScan.service.js:628–642

// ACCURACY: Overlapping text matches from multiple sources should be merged before
// computing display coverage. The same characters matched by 2 sources should count once.
// Use interval-merge algorithm on textMatches before rendering highlights.

// ACCURACY: Sources with only semantic similarity (no lexical match) can appear in
// results if semanticSimilarity > threshold. These will have similarity ≈ 0 but
// a non-zero finalScore. Label these as "Semantic match only" in the UI.
// See: archivePlagiarismScan.service.js:496

// ACCURACY: Quote/bibliography exclusion happens in the backend fingerprinting.

intervalMerge.js — Implement this utility
js/**
 * Merges overlapping intervals, keeping the highest similarity value.
 * Input: [{startIndex, endIndex, sourceId, similarity}]
 * Output: [{startIndex, endIndex, sourceId, similarity}] — no overlaps
 */
export function mergeIntervals(matches) {
  if (!matches.length) return [];
  const sorted = [...matches].sort((a, b) => a.startIndex - b.startIndex);
  const merged = [{ ...sorted[0] }];
  for (let i = 1; i < sorted.length; i++) {
    const last = merged[merged.length - 1];
    const curr = sorted[i];
    if (curr.startIndex <= last.endIndex) {
      last.endIndex = Math.max(last.endIndex, curr.endIndex);
      if (curr.similarity > last.similarity) {
        last.similarity = curr.similarity;
        last.sourceId = curr.sourceId;
      }
    } else {
      merged.push({ ...curr });
    }
  }
  return merged;
}

highlightBuilder.js — Implement this utility
js/**
 * Splits extractedText into annotated segments for rendering.
 * Returns: [{text, highlighted, sourceId, similarity}]
 */
export function buildSegments(extractedText, mergedIntervals) {
  const segments = [];
  let cursor = 0;
  for (const interval of mergedIntervals) {
    if (cursor < interval.startIndex) {
      segments.push({ text: extractedText.slice(cursor, interval.startIndex), highlighted: false });
    }
    segments.push({
      text: extractedText.slice(interval.startIndex, interval.endIndex),
      highlighted: true,
      sourceId: interval.sourceId,
      similarity: interval.similarity,
    });
    cursor = interval.endIndex;
  }
  if (cursor < extractedText.length) {
    segments.push({ text: extractedText.slice(cursor), highlighted: false });
  }
  return segments;
}

❌ Don't average sources[].finalScore to compute overall similarity — use overallSimilarity from the API
❌ Don't show similarity (lexical only) as the headline score per source — use finalScore
❌ Don't highlight quoted/bibliography text — skip those regions visually
❌ Don't show a toast for scan errors — show inline in the upload card
❌ Don't re-sort sources by lexical score by default — default to finalScore descending
❌ Don't ignore semantic-only sources — label them explicitly in the source list
